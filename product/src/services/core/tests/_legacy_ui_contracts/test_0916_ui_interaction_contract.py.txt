def _css_block(styles: str, selector: str) -> str:
    match = re.search(re.escape(selector) + r"\s*\{(.*?)\}", styles, re.S)
    assert match is not None, f"missing css selector: {selector}"
    return match.group(1)

def test_header_back_button_reuses_existing_return_context(script: str) -> None:
    assert script.count("async function loadHistory(") == 1
    assert "#result-header-back" in script
    assert "function returnToResultParent()" in script
    controls = script.split("function updateReturnControls()", 1)[1].split("\n}\n", 1)[0]
    assert 'headerBack.setAttribute("aria-label", label);' in controls
    assert 'headerBack.hidden = state.currentView !== "result";' in controls
    assert ': "返回知识库";' in controls
    # 页头图标按钮不得被文本标签覆盖，否则 SVG 会被清掉
    assert "headerBack.textContent" not in controls
    # 页头、底部与既有入口必须共用同一恢复函数，不得出现第二套逻辑
    assert (
        'document.querySelector("#result-footer-back")?.addEventListener("click", returnToResultParent);'
        in script
    )
    assert script.count("const source = state.returnContext?.sourceView;") == 1
    assert ".result-header-back" in (UI_DIRECTORY / "style.css").read_text(encoding="utf-8")
