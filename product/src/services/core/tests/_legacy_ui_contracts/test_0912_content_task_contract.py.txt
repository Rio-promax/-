def test_0912_paused_jobs_are_not_waiting_or_requeued() -> None:
    client = TestClient(create_app("sqlite:///:memory:"))
    content = client.post(
        "/api/v1/contents",
        json={"kind": "article", "title": "暂停语义", "platform": "web"},
    ).json()
    run = client.post(
        f"/api/v1/contents/{content['id']}/analysis-runs",
        json={"model": "test-model"},
    ).json()
    job_id = run["job"]["id"]

    stored = client.app.state.store.get_job(job_id)
    stored.claim("stale-worker")
    client.app.state.store.save_job(stored)
    stale_attempt = deepcopy(stored)

    paused = client.post(f"/api/v1/jobs/{job_id}/pause")
    assert paused.status_code == 200
    assert paused.json()["execution_state"] == "paused"
    assert paused.json()["queue_position"] is None
    queue = client.get("/api/v1/queue").json()
    assert job_id not in queue["job_ids"]
    assert queue["stats"]["paused"] == 1
    with pytest.raises(StaleJobAttemptError):
        client.app.state.store.save_job(stale_attempt)

    resumed = client.post(f"/api/v1/jobs/{job_id}/resume")
    assert resumed.status_code == 200
    assert resumed.json()["state"] == "queued"


def test_0912_jobs_endpoint_caps_limit_and_orders_terminal_fallback_keys() -> None:
    client = TestClient(create_app("sqlite:///:memory:"))
    jobs = []
    for index in range(3):
        content = client.post(
            "/api/v1/contents",
            json={"kind": "article", "title": f"排序内容 {index}", "platform": "web"},
        ).json()
        run = client.post(
            f"/api/v1/contents/{content['id']}/analysis-runs",
            json={"model": "test-model"},
        ).json()
        jobs.append(client.app.state.store.get_job(run["job"]["id"]))

    now = datetime.now(UTC)
    for index, job in enumerate(jobs):
        job.state = JobState.COMPLETED
        job.completed_at = now - timedelta(minutes=index)
        client.app.state.store.save_job(job)

    response = client.get("/api/v1/jobs", params={"limit": 200})
    assert response.status_code == 200
    returned = [item["id"] for item in response.json()]
    assert returned[:3] == [job.id for job in jobs]
    assert client.get("/api/v1/jobs", params={"limit": 201}).status_code == 422


def test_0912_running_pause_releases_next_job_without_requeueing_paused_job() -> None:
    first = Job(type="analyze", analysis_run_id="run-first")
    second = Job(type="analyze", analysis_run_id="run-second")
    jobs = {first.id: first, second.id: second}

    class Store:
        def get_job(self, job_id):
            return jobs[job_id]

        def save_job(self, job):
            jobs[job.id] = job
            return job

    queue = LocalJobQueue(Store())
    started = Event()
    release = Event()
    execution: list[str] = []

    def runner(job_id, _transcript):
        execution.append(job_id)
        if job_id == first.id:
            job = jobs[job_id]
            job.claim("local-worker")
            started.set()
            release.wait(timeout=3)

    thread = Thread(target=lambda: queue.submit(first.id, None, runner), daemon=True)
    thread.start()
    assert started.wait(timeout=3)
    queue.submit(second.id, None, runner)

    first.transition(JobState.PAUSED)
    jobs[first.id] = first
    queue.remove(first.id)
    release.set()
    thread.join(timeout=3)

    assert execution == [first.id, second.id]
    assert jobs[first.id].state is JobState.PAUSED
    assert queue.snapshot()["job_ids"] == []
