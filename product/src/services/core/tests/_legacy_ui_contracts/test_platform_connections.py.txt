def test_bilibili_qr_login_saves_product_session_and_can_disconnect(monkeypatch) -> None:
    from app.adapters.sources.bilibili import BilibiliQrCode, BilibiliQrPoll

    client = TestClient(create_app("sqlite:///:memory:"))
    monkeypatch.setattr(
        api_main,
        "generate_bilibili_qr_code",
        lambda: BilibiliQrCode("qr-key", "https://passport.bilibili.com/qr", "<svg/>", 180),
    )
    monkeypatch.setattr(
        api_main,
        "poll_bilibili_qr_code",
        lambda key: BilibiliQrPoll(
            "CONNECTED",
            "B站二维码登录成功，正在验证登录会话",
            cookie="SESSDATA=abc; bili_jct=csrf",
            refresh_token="refresh-token",
        ),
    )
    monkeypatch.setattr(
        api_main,
        "probe_platform",
        lambda platform, session_cookie: ("CONNECTED", "测试账号", "B站登录会话可用"),
    )
    saved = {}
    monkeypatch.setattr(
        client.app.state.store,
        "save_bilibili_session",
        lambda cookie, refresh_token: saved.update(cookie=cookie, refresh_token=refresh_token),
    )
    monkeypatch.setattr(
        client.app.state.store,
        "get_bilibili_session_cookie",
        lambda: saved.get("cookie"),
    )

    legacy = client.post(
        "/api/v1/platform-connections/bilibili/connect",
        json={"browser": "chrome", "browser_profile": "Profile 1"},
    )
    assert legacy.status_code == 409

    generated = client.post("/api/v1/platform-connections/bilibili/qrcode")
    assert generated.status_code == 200
    assert generated.json()["qrcode_key"] == "qr-key"
    assert generated.json()["qrcode_svg"] == "<svg/>"

    connected = client.post(
        "/api/v1/platform-connections/bilibili/qrcode/poll",
        json={"qrcode_key": "qr-key"},
    )
    assert connected.status_code == 200
    payload = connected.json()
    assert payload["status"] == "CONNECTED"
    assert payload["account_label"] == "测试账号"
    assert payload["auth_type"] == "qrcode_session"
    assert payload["browser"] is None
    assert saved == {"cookie": "SESSDATA=abc; bili_jct=csrf", "refresh_token": "refresh-token"}
    assert client.get("/api/v1/settings").json()["bilibili_connected"] is True

    disconnected = client.delete("/api/v1/platform-connections/bilibili")
    assert disconnected.status_code == 204
    assert client.get("/api/v1/settings").json()["bilibili_connected"] is False
    assert client.get("/api/v1/platform-connections").json()["connections"][0]["status"] == "NOT_CONNECTED"


def test_bilibili_qr_progress_is_ephemeral_and_does_not_persist_checking(monkeypatch) -> None:
    from app.adapters.sources.bilibili import BilibiliQrCode, BilibiliQrPoll

    client = TestClient(create_app("sqlite:///:memory:"))
    monkeypatch.setattr(
        api_main,
        "generate_bilibili_qr_code",
        lambda: BilibiliQrCode("qr-progress", "https://passport.bilibili.test/qr", "<svg/>", 180),
    )
    results = iter(
        [
            BilibiliQrPoll("WAITING", "等待使用 B站 App 扫码"),
            BilibiliQrPoll("SCANNED", "二维码已扫码，请在 B站 App 中确认登录"),
        ]
    )
    monkeypatch.setattr(api_main, "poll_bilibili_qr_code", lambda key: next(results))

    assert client.post("/api/v1/platform-connections/bilibili/qrcode").status_code == 200
    for expected_message in ("等待使用 B站 App 扫码", "二维码已扫码，请在 B站 App 中确认登录"):
        response = client.post(
            "/api/v1/platform-connections/bilibili/qrcode/poll",
            json={"qrcode_key": "qr-progress"},
        )
        assert response.status_code == 200
        assert response.json()["status"] == CHECKING
        assert response.json()["message"] == expected_message
        assert client.app.state.store.get_platform_connection("bilibili") is None


def test_bilibili_qr_cancel_rejects_a_late_connected_result(monkeypatch) -> None:
    from app.adapters.sources.bilibili import BilibiliQrCode, BilibiliQrPoll

    client = TestClient(create_app("sqlite:///:memory:"))
    monkeypatch.setattr(
        api_main,
        "generate_bilibili_qr_code",
        lambda: BilibiliQrCode("qr-cancel", "https://passport.bilibili.test/qr", "<svg/>", 180),
    )
    saved: list[tuple[str, str | None]] = []

    def late_result(_key: str) -> BilibiliQrPoll:
        client.app.state.bilibili_qr_attempts.clear()
        return BilibiliQrPoll(
            "CONNECTED",
            "B站二维码登录成功，正在验证登录会话",
            cookie="opaque-cookie",
            refresh_token="opaque-refresh",
        )

    monkeypatch.setattr(api_main, "poll_bilibili_qr_code", late_result)
    monkeypatch.setattr(
        client.app.state.store,
        "save_bilibili_session",
        lambda cookie, refresh_token: saved.append((cookie, refresh_token)),
    )

    assert client.post("/api/v1/platform-connections/bilibili/qrcode").status_code == 200
    response = client.post(
        "/api/v1/platform-connections/bilibili/qrcode/poll",
        json={"qrcode_key": "qr-cancel"},
    )
    assert response.status_code == 409
    assert saved == []
    assert client.app.state.store.get_platform_connection("bilibili") is None

def test_bilibili_error_sync_uses_structured_status_only() -> None:
    client = TestClient(create_app("sqlite:///:memory:"))
    store = client.app.state.store
    store.save_platform_connection(
        platform="bilibili",
        status=CONNECTED,
        auth_type="qrcode_session",
        account_label="stable-account",
        login_url="https://passport.bilibili.com/login",
        message="稳定连接",
    )
    source_url = "https://www.bilibili.com/video/BV1example"

    api_main._sync_bilibili_error_status(
        store,
        source_url,
        LinkImportError("接口返回 -101，但未分类"),
    )
    assert store.get_platform_connection("bilibili").status == CONNECTED

    api_main._sync_bilibili_error_status(
        store,
        source_url,
        LinkImportError("接口拒绝", connection_status=RISK_CONTROLLED),
    )
    assert store.get_platform_connection("bilibili").status == RISK_CONTROLLED

    api_main._sync_bilibili_error_status(
        store,
        source_url,
        LinkImportError("网络暂时不可用"),
    )
    assert store.get_platform_connection("bilibili").status == RISK_CONTROLLED

    api_main._sync_bilibili_error_status(
        store,
        source_url,
        LinkImportError("会话失效", connection_status=REAUTH_REQUIRED),
    )
    assert store.get_platform_connection("bilibili").status == REAUTH_REQUIRED

def test_bilibili_session_storage_does_not_partially_replace_a_previous_session(monkeypatch) -> None:
    from app.db import store as store_module
    from app.db.store import SQLiteStore
    from app.platforms.secrets import SecretStorageError

    store = SQLiteStore("sqlite:///:memory:")
    store.set_setting("bilibili_session_cookie", "dpapi:old-cookie")
    store.set_setting("bilibili_refresh_token", "dpapi:old-refresh")

    def protect(value: str, *, purpose: str = "本地凭证") -> str:
        if value == "new-cookie":
            return "dpapi:new-cookie"
        raise SecretStorageError("credential protection blocked")

    monkeypatch.setattr(store_module, "protect_secret", protect)

    with pytest.raises(SecretStorageError):
        store.save_bilibili_session("new-cookie", "new-refresh")

    assert store.get_setting("bilibili_session_cookie") == "dpapi:old-cookie"
    assert store.get_setting("bilibili_refresh_token") == "dpapi:old-refresh"
