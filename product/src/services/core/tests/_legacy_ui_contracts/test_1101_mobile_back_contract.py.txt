"""11-01 手机端壳与 UI 缺陷 —— 源码级契约。

Android 壳的 Java 代码在本机没有单元测试框架（无 Robolectric，且真机行为无法
离机复现），因此这里只断言"关键实现存在/关键缺陷已消失"，真实交互（返回键、
分享弹窗、外链出壳）留给人工在手机上验收。
"""

from __future__ import annotations

import pytest


SERVICE_DIRECTORY = Path(__file__).parents[1]

    SHELL_DIRECTORY / "android" / "app" / "src" / "main" / "java" / "com" / "catsk" / "multisourcekb"

def test_back_bridge_reaches_settings_result_and_batch_views(script: str) -> None:

    block = script[start : start + 900]
    assert 'view === "settings"' in block
    assert 'view === "result"' in block
    assert 'view === "batch-detail"' in block
    assert "returnToResultParent()" in block

def test_mobile_capture_fails_closed_for_non_https_submission() -> None:
    source = _java("HttpPoster.java")

    https_check = 'if (!"https".equalsIgnoreCase(endpoint.getProtocol()))'
    assert https_check in source
    assert source.index(https_check) < source.index("CookieManager.getInstance().getCookie")
    assert source.index(https_check) < source.index('setRequestProperty("Authorization"')
    assert source.index(https_check) < source.index('setRequestProperty("Cookie"')
    assert source.index(https_check) < source.index("output.write(payload)")
    assert "connection.setInstanceFollowRedirects(false)" in source


def test_top_safe_area_is_reserved(styles: str) -> None:
    assert "env(safe-area-inset-top)" in styles
    matches = re.findall(r"\.app-shell\s*\{(.*?)\}", styles, re.S)
    assert matches, "missing .app-shell rule"
    assert any("env(safe-area-inset-top)" in block for block in matches)


def test_settings_tabs_use_two_rows_on_phones(styles: str) -> None:
    matches = re.findall(r"\.settings-subnav\s*\{(.*?)\}", styles, re.S)
    assert matches
    assert any("repeat(4, minmax(0, 1fr))" in block for block in matches)
