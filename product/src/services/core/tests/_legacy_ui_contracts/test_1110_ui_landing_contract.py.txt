def test_r4_three_column_layout(styles: str, markup: str) -> None:
    blocks = _css_blocks(styles, ".ask-layout")
    assert any("200px minmax(0, 1fr) 288px" in b for b in blocks), blocks
    # DOM 层面：范围侧栏必须真的是 ask-layout 的第三个子元素，而不是挂在容器外
    import re as _re
    layout = _re.search(r'<div class="ask-layout">(.*?)\n          </div>', markup, _re.S)
    assert layout, "未找到 ask-layout 容器"

    assert len(children) == 3, children
    assert "ask-scope-panel" in children[2], children
